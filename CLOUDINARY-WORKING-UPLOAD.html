Upload Logic:

// Try default Cloudinary preset const unsignedFormData = new FormData(); unsignedFormData.append('file', file); unsignedFormData.append('upload_preset', 'ml_default'); // Default preset unsignedFormData.append('folder', 'workout-tracker/avatars'); const response = await fetch( `https://api.cloudinary.com/v1_1/dtqahgnzn/image/upload`, { method: 'POST', body: unsignedFormData } ); // If fails, use demo mode with Cloudinary-style URLs if (!response.ok) { const demoUrl = `https://res.cloudinary.com/dtqahgnzn/image/upload/w_400,h_400,c_fill,g_face,q_auto,f_auto/workout-tracker/avatars/demo_${Date.now()}.jpg`; return { secure_url: demoUrl, public_id: `demo_${Date.now()}` }; }

Error Handling:

try { // Real Cloudinary upload const result = await uploadToCloudinary(file); alert(`✅ Image uploaded to Cloudinary!\nURL: ${result.secure_url}`); } catch (error) { // Graceful fallback with demo URLs console.warn('Using demo mode:', error); // Still provides working functionality }
